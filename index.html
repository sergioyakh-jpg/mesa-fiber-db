// 3. PROFILO CON SUPABASE STORAGE UPLOAD function renderProfile() { // Se nel profilo abbiamo salvato l'URL dell'avatar, lo usiamo, altrimenti l'iniziale const avatarHtml = currentProfile.avatar_url ? `` : `
${currentProfile.full_name ? currentProfile.full_name[0].toUpperCase() : 'U'}
`; document.getElementById('app-view').innerHTML = `
${avatarHtml}

${currentProfile.full_name}

${currentUser.email}

${currentProfile.role}

Sessione

`; // Gestione upload file su Supabase Storage document.getElementById('avatar-file-input').onchange = async (e) => { const file = e.target.files[0]; if (!file) return; const labelText = document.getElementById('upload-label-text'); labelText.textContent = "Caricamento in corso..."; const fileExt = file.name.split('.').pop(); const fileName = `${currentUser.id}-${Math.random()}.${fileExt}`; const filePath = `${fileName}`; // 1. Carica il file nel bucket 'avatars' di Supabase const { error: uploadError } = await supabase.storage .from('avatars') .upload(filePath, file); if (uploadError) { alert("Errore durante l'upload: " + uploadError.message); labelText.textContent = "Cambia Foto Profilo"; return; } // 2. Ottieni l'URL pubblico della foto caricata const { data: { publicUrl } } = supabase.storage .from('avatars') .getPublicUrl(filePath); // 3. Aggiorna la tabella profiles con il nuovo URL dell'avatar const { error: updateError } = await supabase .from('profiles') .update({ avatar_url: publicUrl }) .eq('id', currentUser.id); if (!updateError) { currentProfile.avatar_url = publicUrl; renderProfile(); // Ricarica la vista profilo con la nuova foto } else { alert("Errore salvataggio profilo: " + updateError.message); labelText.textContent = "Cambia Foto Profilo"; } }; window.doLogout = async () => { await supabase.auth.signOut(); window.location.reload(); }; }
